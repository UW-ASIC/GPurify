VERSION 5.8 ;
UNITS DATABASE MICRONS 1000 ; END UNITS
LAYER M1 TYPE ROUTING ; DIRECTION HORIZONTAL ; WIDTH 0.01 ; END M1
LAYER cut12 TYPE CUT ; END cut12
LAYER M2 TYPE ROUTING ; DIRECTION VERTICAL ; WIDTH 0.02 ; END M2
VIA VIA12 DEFAULT
  LAYER M1 ; RECT -0.01 -0.01 0.01 0.01 ;
  LAYER cut12 ; RECT -0.005 -0.005 0.005 0.005 ;
  LAYER M2 ; RECT -0.01 -0.01 0.01 0.01 ;
END VIA12
NONDEFAULTRULE wide
  LAYER M1 WIDTH 0.04 ; END M1
END wide
MACRO cell
  CLASS CORE ; ORIGIN 0 0 ; SIZE 0.004 BY 0.002 ;
  PIN a DIRECTION INPUT ; PORT LAYER M1 ; RECT 0 0 0.001 0.001 ; END END a
  OBS LAYER M2 ; RECT 0.003 0.001 0.004 0.002 ; END
END cell
END LIBRARY
VERSION 5.8 ;
DESIGN top ;
UNITS DISTANCE MICRONS 1000 ;
DIEAREA ( 0 0 ) ( 1000 1000 ) ;
ROW r0 core 0 0 N DO 10 BY 1 STEP 4 0 ;
TRACKS X 5 DO 100 STEP 10 LAYER M2 ;
VIAS 1 ;
- v + VIARULE g + CUTSIZE 20 20 + LAYERS M1 cut12 M2 + CUTSPACING 30 30 + ENCLOSURE 20 50 50 20 + ROWCOL 2 3 + PATTERN 1_A_1_E ;
END VIAS
COMPONENTS 2 ;
- u1 cell + PLACED ( 100 200 ) FS ;
- u2 cell + FIXED ( 300 200 ) W ;
END COMPONENTS
PINS 1 ;
- p + NET n + DIRECTION INPUT + PORT + LAYER M1 ( -5 0 ) ( 5 10 ) + PLACED ( 100 0 ) S ;
END PINS
SPECIALNETS 1 ;
- vdd ( * a ) + ROUTED M1 2 + SHAPE STRIPE ( 0 0 ) ( 5 0 ) VIRTUAL ( 7 1 ) RECT ( -3 0 -1 2 ) ( 7 7 ) v DO 2 BY 1 STEP 300 0 ;
END SPECIALNETS
NETS 2 ;
- n ( u1 a ) ( PIN p ) + ROUTED M1 ( 0 50 ) ( 50 * 20 ) VIA12 ( * * 15 ) ( * 0 ) NEW M2 TAPER ( 500 0 ) ( 500 100 ) + USE SIGNAL ;
- w + NONDEFAULTRULE wide + ROUTED M1 ( 0 600 ) ( 400 600 ) ;
END NETS
FILLS 1 ;
- LAYER M2 + OPC RECT ( 800 800 ) ( 900 900 ) ;
END FILLS
END DESIGN
